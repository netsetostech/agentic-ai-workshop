Pub/Sub hands the worker whatever the publisher sent; parsing it into a model at the edge means a malformed message fails here, with a field name, not three functions later with a KeyError. chunk_hash hashes the text with every run of whitespace collapsed to one space and the ends stripped, so a re-wrapped or re-indented paragraph hashes the same and a changed figure is a new chunk. is_stale is the pure half of the generation guard (3.4 reads the ledger half): older than the ledger means a late redelivery; a generation the ledger has not seen, or anything that is not a number, is never stale - it refuses to guess.

_EFFECTIVE_TEXT = re.compile(r"effective[ _-]?(?:from|date)?\s*[:=]\s*(\d{4}-\d{2}-\d{2})", re.I)
_EFFECTIVE_NAME = re.compile(r"_effective_(\d{4}-\d{2}-\d{2})\.")
# ...
def effective_from_of(name: str, text: str | None) -> str | None:
    # ...
    m = _EFFECTIVE_NAME.search(name or "")
    if m:
        return m.group(1)
    if text:
        m = _EFFECTIVE_TEXT.search(text[:3000])
        if m:

A document declares its own effective date; the pipeline never guesses one. The name is checked first (_effective_2026-10-01., for a PDF nobody can edit), then the first 3,000 characters for effective from: 2026-10-01 - case-insensitive, from or date optional, a colon or an equals sign. Revision 2 of the handbook says Effective from: 2026-10-01 on its third line; revision 1 is undated. def parse(project_id: str, processor_id: str, content: bytes,
          mime_type: str) -> tuple[str, int]:
    """Return (text with a form feed between pages, page_count)."""
    client, name = _client_and_name(project_id, processor_id)
    if mime_type == "application/pdf":
        reader = PdfReader(io.BytesIO(content))
        if len(reader.pages) > ONLINE_PAGE_LIMIT:
            pages: list[str] = []
            for first in range(0, len(reader.pages), ONLINE_PAGE_LIMIT):
                writer = PdfWriter()
                for page in reader.pages[first:first + ONLINE_PAGE_LIMIT]:
                    writer.add_page(page)
                buf = io.BytesIO()
                writer.write(buf)
                pages.extend(_process(client, name, buf.getvalue(), mime_type))
            return "\f".join(pages), len(pages)
    pages = _process(client, name, content, mime_type)
    return "\f".join(pages), len(pages)

An online request takes at most 15 pages, for OCR, Layout Parser and Form Parser alike, so pypdf opens the file and, past the limit, a PdfWriter cuts 15-page slices that go up one after another and come back in order, joined with a form feed - the one character that lets the chunker cut page by page and name the page a citation shows. _page_texts() (lines 43 to 76) turns either response shape, OCR's anchored pages or the Layout Parser's document_layout blocks, into one string per page.

What just happened

Three documents, three arithmetic answers, before a rupee is spent. The handbook is text: one page, no processor. The 29-page Code on Wages is 2 slices (15 + 14), about half a minute. shared/pii.py - inspect_many()
def inspect_many(texts: list[str]) -> list[list[dict]]:
    # ...
    out: list[list[dict]] = [[] for _ in texts]
    encoded = [t.encode("utf-8") for t in texts]
    i = 0
    while i < len(encoded):
        starts, parts, size, j = [], [], 0, i
        while j < len(encoded) and (not parts or size + len(SEPARATOR) + len(encoded[j]) <= REQUEST_BYTES):
            if parts:
                size += len(SEPARATOR)
            starts.append(size)
            parts.append(encoded[j])
            size += len(encoded[j])
            j += 1
        data = SEPARATOR.join(parts)
        for f in _inspect_bytes(data):
            start = f.location.byte_range.start if f.location.byte_range else 0
            k = bisect.bisect_right(starts, start) - 1
            out[i + k].append({"info_type": f.info_type.name,
                               "likelihood": f.likelihood.name,
                               "offset": start - starts[k]})
        i = j
    return out

The chunker: sections, windows, locators

Two rules in two files, held to one output by a gate: a section per heading, 2,000-character windows with a 200 overlap cut page by page, and a locator that survives an edit above it.

The chunker lives twice on purpose: shared/documind_corpus.py is the notebooks' copy, pasted verbatim into 4.2 and 4.5, and main.py is the worker's; both cut the same bytes into the same texts, locators and hashes, so a corpus seeded from Colab and the same file uploaded to the bucket are one set of chunks.

The rules, in the notebooks' copy

shared/documind_corpus.py - the constants, windows() and chunk_hash()
CHUNK_CHARS, CHUNK_OVERLAP = 2000, 200                     # services/ingest/main.py
# ...
_SECTION = re.compile(r"^## +(.+?) *$", re.M)
_CODE = re.compile(r"^([A-Z][A-Z0-9]{0,7}(?:-[A-Z0-9]{1,6}){1,2})\b")   # NP-03, IT-SEC-04, MSA-04, GEN-014
# ...
def windows(text: str) -> list:
    # ...
    text = text.strip()
    out, start = [], 0
    while start < len(text):
        end = min(len(text), start + CHUNK_CHARS)
        if end < len(text):
            cut = text.rfind(". ", start + CHUNK_CHARS // 2, end)
            if cut != -1:
                end = cut + 1
        piece = text[start:end].strip()
        if piece:
            out.append(piece)
        if end >= len(text):
            break
        start = max(end - CHUNK_OVERLAP, start + 1)
    return out
# ...
def chunk_hash(text: str) -> str:
    # ...
    return hashlib.sha256(re.sub(r"\s+", " ", text).strip().encode("utf-8")).hexdigest()

def chunk_document(doc: dict, tenant: str, version: str = "") -> list:
    # ...
    text = re.sub(r"\A\s*<!--.*?-->\s*", "", doc["text"], count=1, flags=re.S)   # a mirror's provenance header
    base = {"tenant_id": tenant, "source_uri": doc["source_uri"], "doc_type": doc["doc_type"], "kind": "text"}
    at = f"@{version}" if version else ""
    out = []
    heads = list(_SECTION.finditer(text))
    if heads:                                                  # a handbook: one chunk per section
        # ...
        for n, h in enumerate(heads):
            body = text[h.end(): heads[n + 1].start() if n + 1 < len(heads) else len(text)].strip()
            title = h.group(1).strip()
            code = _CODE.match(title)
            key = code.group(1) if code else f"s{n + 1}"
            for k, piece in enumerate(windows(f"{title}\n{body}")):
                loc = key + (f"-{k}" if k else "")
                out.append({**base, "chunk_id": f"{tenant}:{doc['slug']}{at}#{loc}", "text": piece, "page_start": 1,
                            "section": title, "locator": loc, "chunk_hash": chunk_hash(piece)})

windows takes 2,000 characters, looks back over the second half for a sentence end (". ") and cuts there when it finds one, then starts the next window 200 characters before the cut. _SECTION is a line starting with ## ; _CODE is a clause code at the start of a heading - NP-03, IT-SEC-04, MSA-04. chunk_document drops a mirror's provenance header first (the leading <!-- --> that fetch_real.py writes). A document with headings is a handbook: the text above the first heading is preamble, and each section is one chunk keyed by its code, or s4 when the heading has none, windowed within itself only past 2,000 characters. Anything else is a PDF mirror (lines 139 to 145): pages split on the form feed, windows per page, locator p7-1 for the second window on page 7. The id carries the tenant, the slug, the version on a re-issue (@1f3a9c2b) and the locator.

The same rules in the worker

services/ingest/main.py - CHUNK_CHARS and _chunk()
CHUNK_CHARS, CHUNK_OVERLAP = 2000, 200
# ...
_SECTION = re.compile(r"^## +(.+?) *$", re.M)
_CODE = re.compile(r"^([A-Z][A-Z0-9]{0,7}(?:-[A-Z0-9]{1,6}){1,2})\b")   # NP-03, IT-SEC-04, MSA-04, GEN-014
# ...
def _chunk(text: str) -> list[dict]:
    # ...
    text = re.sub(r"\A\s*<!--.*?-->\s*", "", text, count=1, flags=re.S)
    heads = list(_SECTION.finditer(text.strip()))
    out = []
    if heads:
        text = text.strip()
        # ...
        for n, h in enumerate(heads):
            title = h.group(1).strip()
            body = text[h.end(): heads[n + 1].start() if n + 1 < len(heads) else len(text)].strip()
            code = _CODE.match(title)
            key = code.group(1) if code else f"s{n + 1}"
            for k, (_, piece) in enumerate(_windows(f"{title}\n{body}")):
                out.append({"text": piece, "kind": "text", "page_start": None,
                            "locator": key + (f"-{k}" if k else ""), "section": title})
    else:
        paged = "\f" in text
        for p, page in enumerate(text.split("\f"), 1):
            for k, (_, piece) in enumerate(_windows(page)):
                out.append({"text": piece, "kind": "text", "page_start": p if paged else None,
                            "locator": f"p{p}-{k}" if paged else f"w{k}", "section": None})
    for c in out:
        c["chunk_hash"] = chunk_hash(c["text"])
    return out

_windows() (line 155) is the loader's loop line for line, returning each window's start offset as well. _chunk adds what the loader does not need: text with no form feed gets w0, w1 and no page - None is more honest than 1 - and every chunk carries kind: "text", the shared contract's word beside figure and segment. services/ingest/indexer.py - batches()
def batches(texts: list[str]) -> list[list[str]]:
    out, cur, cur_tokens = [], [], 0
    for text in texts:
        tokens = max(1, len(text) // CHARS_PER_TOKEN)
        if cur and (len(cur) >= EMBED_BATCH or cur_tokens + tokens > EMBED_TOKENS):
            out.append(cur)
            cur, cur_tokens = [], 0
        cur.append(text)
        cur_tokens += tokens
    if cur:
        out.append(cur)
    return out

def embed_all(texts: list[str]) -> list[list[float]]:
    out: list[list[float]] = []
    for batch in batches(texts):
        response = _embed.models.embed_content(
            model=EMBEDDING_MODEL,
            contents=batch,
            config={"output_dimensionality": EMBEDDING_DIMENSIONS, "task_type": EMBEDDING_TASK_TYPE},
        )
        vectors = [list(embedding.values) for embedding in (response.embeddings or [])]
        if len(vectors) != len(batch) or any(not valid_vector(v) for v in vectors):
            raise ValueError("Embedding response must contain one finite 768-dimensional vector per text")
        out.extend(vectors)
    return out

def valid_vector(vector) -> bool:
    return (vector is not None and len(list(vector)) == EMBEDDING_DIMENSIONS
            and all(math.isfinite(v) for v in vector))
# ...
def document_embedding_matches(row: dict) -> bool:
    """Unstamped legacy vectors used the API default (RETRIEVAL_QUERY), not document encoding."""
    return (row.get("embedding_model") == EMBEDDING_MODEL
            and str(row.get("embedding_version")) == str(EMBEDDING_VERSION)
            and row.get("embedding_task_type") == EMBEDDING_TASK_TYPE
            and valid_vector(row.get("embedding")))

A batch closes at 250 texts or when the next text would push it past 15,000 estimated tokens; a request over either limit fails whole, which is how the first live load failed every long Act on 6 September 2026. embed_all refuses a response that is not one finite 768-dimensional vector per text - an empty list, a short vector, a NaN, an infinity - and commands/tests/test_document_embeddings.py pins each case with a stubbed SDK. document_embedding_matches asks four questions of a stored row, and exists because of a defect found on 17 September 2026: the worker used to omit the task type, and the API's default for an omitted task is RETRIEVAL_QUERY. Documents must be encoded as documents; the API embeds questions as RETRIEVAL_QUERY (services/rag-api/retriever.py). A row without the stamp was encoded the wrong way, so the carry-over never reuses it, current_chunks never adopts it, and the repair in step 8 regenerates it.

    vectors, misses = [], []
    for index, chunk in enumerate(chunks):
        vector = held.get(chunk.get("chunk_hash") or "")
        vectors.append(vector)
        if vector is None:
            misses.append(index)
    return vectors, misses

def embed_with_carry_over(db: firestore.Client, doc, chunks: list[dict]) -> tuple[list[list[float]], dict]:
    held = held_vectors(db, doc)
    vectors, misses = plan_carry_over(chunks, held)
    fresh = embed_all([chunks[index]["text"] for index in misses]) if misses else []
    for index, vector in zip(misses, fresh):
        vectors[index] = vector
    return vectors, {"reused": len(chunks) - len(misses), "embedded": len(misses)}

shared/sparse_encoder.py - sparse_encode()
SPARSE_DIMS = 1 << 20
SPARSE_ENCODER_VERSION = "blake2b-tf-v1"
# ...
def _tokens(text: str) -> list[str]:
    return re.findall(r"[a-z0-9\-]+", (text or "").lower())
# ...
def sparse_encode(text: str) -> tuple[list[float], list[int]]:
    # ...
    per_dimension: dict[int, float] = {}
    for token, count in Counter(_tokens(text)).items():
        dimension = int(hashlib.blake2b(token.encode(), digest_size=4).hexdigest(), 16) % SPARSE_DIMS
        weight = 1.0 + (count - 1) * 0.5
        per_dimension[dimension] = per_dimension.get(dimension, 0.0) + weight
    dimensions = sorted(per_dimension)
    return [per_dimension[d] for d in dimensions], dimensions

The one rule to remember

Nothing on this path decides by name or by time, and nothing guesses: the tenant is the folder, the version is the hash of the bytes, the chunk is the hash of its whitespace-collapsed text plus a locator that survives an edit above it, a vector is reused only when its stamp says what made it, and every row, datapoint and insert carries the same id so a retry lands on itself.
