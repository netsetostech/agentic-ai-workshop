rag_load_project_billing() { : "${PROJECT:?Restore the intended project ID}" "${DEMO_ROOT:?Restore DEMO_ROOT}" || return unset BILLING_ACCOUNT_NAME BILLING_ACCOUNT_ID TF_VAR_billing_account_id BILLING_ACCOUNT_ID="$(python - "$PROJECT" "$DEMO_ROOT" <<'RAG_BILLING' import json import re import subprocess import sys import tempfile from pathlib import Path project, root = sys.argv[1:] flags = [f"--project={project}", f"--billing-project={project}", "--format=json"] def read_gcloud(args): try: return json.loads(subprocess.check_output(["gcloud", *args, *flags], text=True)) except (subprocess.CalledProcessError, json.JSONDecodeError) as exc: raise SystemExit(f"STOP: cannot verify project billing; no inputs were saved ({type(exc).__name__}).") info = read_gcloud(["projects", "describe", project]) if info.get("projectId") != project or info.get("lifecycleState") != "ACTIVE": raise SystemExit("STOP: PROJECT must identify the intended ACTIVE project.") billing = read_gcloud(["billing", "projects", "describe", project]) if billing.get("projectId") != project or billing.get("billingEnabled") is not True: raise SystemExit("STOP: billing is not enabled for the intended project.") match = re.fullmatch(r"billingAccounts/([0-9A-Za-z]{6}-[0-9A-Za-z]{6}-[0-9A-Za-z]{6})", billing.get("billingAccountName") or "") if not match or match.group(1) == "000000-000000-000000": raise SystemExit("STOP: the linked billing account ID is missing or invalid.") account = match.group(1).upper() target = Path(root) / "terraform" / "runbook-inputs.tfvars.json" if not target.parent.is_dir(): raise SystemExit("STOP: restore DEMO_ROOT to the existing deployment directory.") try: values = json.loads(target.read_text()) if target.exists() else {} except (OSError, json.JSONDecodeError): raise SystemExit("STOP: inspect the existing runbook-inputs.tfvars.json; it was not replaced.") if not isinstance(values, dict): raise SystemExit("STOP: runbook-inputs.tfvars.json must contain a JSON object.") if values.get("project_id") not in (None, "", project): raise SystemExit("STOP: saved inputs belong to another project; verify the state/backend before proceeding.") values.update(project_id=project, billing_account_id=account) with tempfile.NamedTemporaryFile(mode="w", dir=target.parent, prefix=".rag-inputs-", delete=False) as f: json.dump(values, f, indent=2) f.write("\n") temporary = Path(f.name) try: temporary.replace(target) finally: temporary.unlink(missing_ok=True) print(account) RAG_BILLING )" || return BILLING_ACCOUNT_NAME="billingAccounts/$BILLING_ACCOUNT_ID" export BILLING_ACCOUNT_NAME BILLING_ACCOUNT_ID export TF_VAR_billing_account_id="$BILLING_ACCOUNT_ID" printf 'PASS: verified active project %s and linked billing account %s.\n' "$PROJECT" "$BILLING_ACCOUNT_ID" printf 'Saved Terraform inputs: %s/terraform/runbook-inputs.tfvars.json\n' "$DEMO_ROOT" } rag_load_project_billing
